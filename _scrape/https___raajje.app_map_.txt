        // ============ INDIVIDUAL PLOT SIZE CALCULATION ============
        var PLOT_TOL = 1e-6;
        var PLOT_SNAP_DIST = 5.0; // meters - snap plot line endpoints to parcel boundary within this distance

        function pEq(a, b) {
            return Math.abs(a[0] - b[0]) < PLOT_TOL && Math.abs(a[1] - b[1]) < PLOT_TOL;
        }

        // Find closest point on segment segA-segB to point pt
        function closestPointOnSeg(pt, segA, segB) {
            var dx = segB[0] - segA[0], dy = segB[1] - segA[1];
            var lenSq = dx * dx + dy * dy;
            if (lenSq < 1e-12) return { point: [segA[0], segA[1]], dist: Math.sqrt((pt[0]-segA[0])*(pt[0]-segA[0])+(pt[1]-segA[1])*(pt[1]-segA[1])) };
            var t = Math.max(0, Math.min(1, ((pt[0] - segA[0]) * dx + (pt[1] - segA[1]) * dy) / lenSq));
            var proj = [segA[0] + t * dx, segA[1] + t * dy];
            var dist = Math.sqrt((pt[0]-proj[0])*(pt[0]-proj[0])+(pt[1]-proj[1])*(pt[1]-proj[1]));
            return { point: proj, dist: dist };
        }

        // Snap a point to the nearest parcel boundary edge if close enough
        function snapToBoundary(pt, boundarySegs) {
            var best = null;
            for (var i = 0; i < boundarySegs.length; i++) {
                var result = closestPointOnSeg(pt, boundarySegs[i][0], boundarySegs[i][1]);
                if (!best || result.dist < best.dist) best = result;
            }
            if (best && best.dist < PLOT_SNAP_DIST) return best.point;
            return pt;
        }

        function pSegInt(a1, a2, b1, b2) {
            var d1x = a2[0] - a1[0], d1y = a2[1] - a1[1];
            var d2x = b2[0] - b1[0], d2y = b2[1] - b1[1];
            var cr = d1x * d2y - d1y * d2x;
            if (Math.abs(cr) < PLOT_TOL) return null;
            var t = ((b1[0] - a1[0]) * d2y - (b1[1] - a1[1]) * d2x) / cr;
            var u = ((b1[0] - a1[0]) * d1y - (b1[1] - a1[1]) * d1x) / cr;
            if (t > PLOT_TOL && t < 1 - PLOT_TOL && u > PLOT_TOL && u < 1 - PLOT_TOL) {
                return [a1[0] + t * d1x, a1[1] + t * d1y];
            }
            return null;
        }

        function pInPoly(pt, poly) {
            var c = false;
            for (var i = 0, j = poly.length - 1; i < poly.length; j = i++) {
                if ((poly[i][1] > pt[1]) !== (poly[j][1] > pt[1]) &&
                    pt[0] < (poly[j][0] - poly[i][0]) * (pt[1] - poly[i][1]) / (poly[j][1] - poly[i][1]) + poly[i][0]) {
                    c = !c;
                }
            }
            return c;
        }

        function pArea(coords) {
            var n = coords.length;
            if (n < 3) return 0;
            var a = 0;
            for (var i = 0; i < n; i++) {
                var j = (i + 1) % n;
                a += coords[i][0] * coords[j][1] - coords[j][0] * coords[i][1];
            }
            return Math.abs(a) / 2;
        }

        function calculateIndividualPlotSizes(parcelGeometry, plotLineFeatures, addressFeatures) {

                }

                var segs = [];
                for (var i = 0; i < boundarySegs.length; i++) {
                    segs.push([boundarySegs[i][0], boundarySegs[i][1]]);
                }

                // Process plot lines - snap endpoints to parcel boundary
                plotLineFeatures.forEach(function(f) {
                    var g = f.getGeometry();
                    var lines = g.getType() === 'MultiLineString' ? g.getCoordinates() : [g.getCoordinates()];
                    lines.forEach(function(lc) {
                        // Snap each endpoint to boundary if close
                        var snapped = [];
                        for (var i = 0; i < lc.length; i++) {
                            snapped.push(snapToBoundary(lc[i], boundarySegs));
                        }
                        for (var i = 0; i < snapped.length - 1; i++) {
                            if (!pEq(snapped[i], snapped[i + 1])) {
                                segs.push([snapped[i], snapped[i + 1]]);
                            }
                        }
                    });
                });

                if (segs.length < 3) return null;

                var pts = [];
                function addPt(p) {
                    for (var i = 0; i < pts.length; i++) {
                        if (pEq(pts[i], p)) return i;
                    }
                    pts.push([p[0], p[1]]);
                    return pts.length - 1;
                }

                segs.forEach(function(s) { addPt(s[0]); addPt(s[1]); });

                // Check if point lies on segment (collinear and within bounds)
                function ptOnSeg(p, a, b) {
                    var cross = (b[0] - a[0]) * (p[1] - a[1]) - (b[1] - a[1]) * (p[0] - a[0]);
                    if (Math.abs(cross) > PLOT_TOL) return false;
                    var dot = (p[0] - a[0]) * (b[0] - a[0]) + (p[1] - a[1]) * (b[1] - a[1]);
                    if (dot < -PLOT_TOL) return false;
                    var lenSq = (b[0] - a[0]) * (b[0] - a[0]) + (b[1] - a[1]) * (b[1] - a[1]);
                    return dot <= lenSq + PLOT_TOL;
                }

                var subSegs = [];
                for (var i = 0; i < segs.length; i++) {
                    var intPts = [segs[i][0], segs[i][1]];
                    for (var j = 0; j < segs.length; j++) {
                        if (i === j) continue;
                        // Check crossing intersection
                        var ip = pSegInt(segs[i][0], segs[i][1], segs[j][0], segs[j][1]);
                        if (ip) { intPts.push(ip); continue; }
                        // Check if endpoint of seg j lies on seg i (for snapped points)
                        if (ptOnSeg(segs[j][0], segs[i][0], segs[i][1])) intPts.push(segs[j][0]);
                        if (ptOnSeg(segs[j][1], segs[i][0], segs[i][1])) intPts.push(segs[j][1]);
                    }

                    var dx = segs[i][1][0] - segs[i][0][0];
                    var dy = segs[i][1][1] - segs[i][0][1];
                    var len = Math.sqrt(dx * dx + dy * dy);
                    if (len < PLOT_TOL) continue;

                    intPts.sort(function(a, b) {
                        var ta = ((a[0] - segs[i][0][0]) * dx + (a[1] - segs[i][0][1]) * dy) / (len * len);
                        var tb = ((b[0] - segs[i][0][0]) * dx + (b[1] - segs[i][0][1]) * dy) / (len * len);
                        return ta - tb;
                    });

                    for (var k = 0; k < intPts.length - 1; k++) {
                        if (!pEq(intPts[k], intPts[k + 1])) {
                            var i1 = addPt(intPts[k]);
                            var i2 = addPt(intPts[k + 1]);
                            if (i1 !== i2) subSegs.push([i1, i2]);
                        }
                    }
                }

                var uSet = {}, uSegs = [];
                subSegs.forEach(function(s) {
                    var k = Math.min(s[0], s[1]) + '-' + Math.max(s[0], s[1]);
                    if (!uSet[k]) { uSet[k] = true; uSegs.push(s); }
                });

                if (uSegs.length < 3 || pts.length < 3) return null;

                var adj = {};
                uSegs.forEach(function(s) {
                    if (!adj[s[0]]) adj[s[0]] = [];
                    if (!adj[s[1]]) adj[s[1]] = [];
                    adj[s[0]].push({ v: s[1], a: pAngle(pts[s[0]], pts[s[1]]) });
                    adj[s[1]].push({ v: s[0], a: pAngle(pts[s[1]], pts[s[0]]) });
                });
                for (var v in adj) {
                    adj[v].sort(function(x, y) { return x.a - y.a; });
                }

                var used = {}, faces = [];
                for (var v in adj) {
                    var vi = parseInt(v);
                    adj[vi].forEach(function(e) {
                        var k = vi + '-' + e.v;
                        if (used[k]) return;
                        var face = [], f = vi, t = e.v, iter = 0;
                        while (iter++ < pts.length * 3) {
                            used[f + '-' + t] = true;
                            face.push(f);
                            var nb = adj[t];
                            if (!nb) break;
                            var nx = -1;
                            for (var i = 0; i < nb.length; i++) {
                                if (nb[i].v === f) {
                                    nx = nb[(i + 1) % nb.length].v;
                                    break;
                                }
                            }
                            if (nx === -1) break;
                            f = t;
                            t = nx;
                            if (f === vi && t === e.v) break;
                        }
                        if (face.length >= 3) {
                            var fc = face.map(function(idx) { return pts[idx]; });
                            var a = pArea(fc);
                            if (a > PLOT_TOL) faces.push({ c: fc, a: a });
                        }
                    });
                }

                if (faces.length === 0) return null;

                var maxA = 0, oi = -1;
                faces.forEach(function(f, i) {
                    if (f.a > maxA) { maxA = f.a; oi = i; }
                });
                var inner = faces.filter(function(f, i) { return i !== oi; });

                if (inner.length === 0) {
                    inner = [{ c: coords.slice(0, -1), a: pArea(coords.slice(0, -1)) }];
                }

                var plots = inner.map(function(f) {
                    return { coords: f.c, areaSqM: f.a, areaSqFt: Math.round(f.a * 10.7639), addrs: [] };
                });

                addressFeatures.forEach(function(af) {
                    var pt = af.feature.getGeometry().getCoordinates();
                    for (var i = 0; i < plots.length; i++) {
                        if (pInPoly(pt, plots[i].coords)) {
                            plots[i].addrs.push(af.name);
                            break;
                        }
                    }
                });

                var unassignedAddrs = addressFeatures.filter(function(af) {
                    var pt = af.feature.getGeometry().getCoordinates();
                    return !plots.some(function(p) { return p.coords.length > 0 && pInPoly(pt, p.coords); });
                });
                if (unassignedAddrs.length > 0) {
                    var parcelArea = pArea(coords.slice(0, -1));
                    var assignedArea = plots.reduce(function(sum, p) { return sum + p.areaSqM; }, 0);
                    var remainArea = Math.max(0, parcelArea - assignedArea);
                    plots.push({ coords: [], areaSqM: remainArea, areaSqFt: Math.round(remainArea * 10.7639), addrs: unassignedAddrs.map(function(a) { return a.name; }) });
                }

                return plots;
            } catch (e) {
                console.error('Plot size calculation error:', e);
                return null;
            }
        }

        // Load data from MySQL API
        function loadDataFromAPI() {
            // Small layers loaded fully, large layers loaded on-demand by viewport
            var smallLayers = [
                { name: 'Atoll Boundary', key: 'atoll_boundaries', varName: 'json_AtollBoundaryLine_20' },
                { name: 'Admin Atoll', key: 'administrative_atolls', varName: 'json_AdministrativeAtoll_21' },
                { name: 'Airports', key: 'airports', varName: 'json_Airport_10' },
                { name: 'Capitals', key: 'atoll_capitals', varName: 'json_AtollCapital_13' },
                { name: 'Island Names', key: 'island_names', varName: 'json_IslandName_23' }
            ];
            
            // Large layers - initialize as empty, will load on-demand
            var largeLayers = [
                { name: 'Parcels', key: 'parcels', varName: 'json_ParcelCategory_5' },
                { name: 'House Parcels', key: 'house_parcels', varName: 'json_HouseParcels' },
                { name: 'Plot Lines', key: 'plot_lines', varName: 'json_plotLine_8' },
                { name: 'Addresses', key: 'addresses', varName: 'json_Address_24' }
            ];
            
            var totalSteps = smallLayers.length + largeLayers.length + 2;
            var loadIndex = 0;
            
            function updateProgress(step, text) {
                var percentage = Math.round((step / totalSteps) * 100);
                updateLoadingProgress(percentage, text);
            }
            
            // Initialize large layers as empty (will load on-demand)
            largeLayers.forEach(function(layer, i) {
                updateProgress(i + 1, 'Preparing ' + layer.name + '...');
                var emptyGeoJSON = { type: 'FeatureCollection', features: [] };
                layerData[layer.key] = emptyGeoJSON;
                window[layer.varName] = emptyGeoJSON;
            });
            
            // Load small layers in PARALLEL for much faster startup
            updateProgress(largeLayers.length, 'Loading layers in parallel...');
            
            var layerPromises = smallLayers.map(function(layer, index) {
                return fetch(API_BASE_URL + '?layer=' + layer.key)
                    .then(function(response) {
                        if (!response.ok) {
                            throw new Error('HTTP error! status: ' + response.status);
                        }
                        return response.json();
                    })
                    .then(function(data) {
                        layerData[layer.key] = data;
                        window[layer.varName] = data;
                        updateProgress(largeLayers.length + index + 1, 'Loaded ' + layer.name + '...');
                        return { success: true, layer: layer.name };
                    })
                    .catch(function(error) {
                        console.warn('Failed to load layer from API: ' + layer.key, error);
                        layerData[layer.key] = { type: 'FeatureCollection', features: [] };
                        window[layer.varName] = layerData[layer.key];
                        updateProgress(largeLayers.length + index + 1, 'Skipped ' + layer.name + '...');
                        return { success: false, layer: layer.name, error: error };
                    });
            });
            
            // Wait for all layers to load in parallel
            Promise.all(layerPromises)
                .then(function(results) {
                    updateProgress(totalSteps - 1, 'Initializing map...');
                    setTimeout(function() {
                        initializeMapWithAPI();
                    }, 100);
                })
                .catch(function(error) {
                    console.error('Error loading layers:', error);
                    // Continue anyway with empty layers
                    updateProgress(totalSteps - 1, 'Initializing map...');
                    setTimeout(function() {
                        initializeMapWithAPI();
                    }, 100);
                });
            
            // Start loading
            updateProgress(0, 'Connecting to database...');
        }
        
        // Load data from JavaScript files (legacy mode)
        function loadDataFromFiles() {
            var layerFiles = [
                { name: 'Parcels', src: './assets/ParcelCategory_5.js' },
                { name: 'House Parcels', src: './assets/HouseParcels.js' },
                { name: 'Plot Lines', src: './assets/plotLine_8.js' },
                { name: 'Atoll Boundary', src: './assets/AtollBoundaryLine_20.js' },
                { name: 'Admin Atoll', src: './assets/AdministrativeAtoll_21.js' },
                { name: 'Airports', src: './assets/Airport_10.js' },
                { name: 'Capitals', src: './assets/AtollCapital_13.js' },
                { name: 'Island Names', src: './assets/IslandName_23.js' },
                { name: 'Addresses', src: './assets/Address_24_filtered.js' }
            ];
            
            var totalSteps = layerFiles.length + 2;
            var loadIndex = 0;
            
            function updateProgress(step, text) {
                var percentage = Math.round((step / totalSteps) * 100);
                updateLoadingProgress(percentage, text);
            }
            
            function loadNext() {
                if (loadIndex < layerFiles.length) {
                    var file = layerFiles[loadIndex];
                    updateProgress(loadIndex + 1, 'Loading ' + file.name + '...');
                    
                    var script = document.createElement('script');
                    script.src = file.src;
                    script.onload = function() {
                        loadIndex++;
                        loadNext();
                    };
                    script.onerror = function() {
                        console.warn('Failed to load: ' + file.src);
                        loadIndex++;
                        loadNext();
                    };
                    document.head.appendChild(script);
                } else {
                    updateProgress(totalSteps - 1, 'Initializing map...');
                    setTimeout(function() {
                        initializeMap();
                    }, 100);
                }
            }
            
            // Start loading
            updateProgress(0, 'Starting...');
            loadNext();
        }

            function loadAddressesFromAPI() {
                var view = map.getView();

                addressLoadingBbox = bbox;

                if (addressAbortController) {
                    addressAbortController.abort();
                }
                
                // Create new AbortController for this request
                addressAbortController = new AbortController();
                
                fetch(API_BASE_URL + '?layer=addresses&bbox=' + bbox, {

                            addressSource.addFeatures(features);
                            allAddressFeatures = features;
                            window.allAddressFeatures = features;

                        addressAbortController = null;
                    })
                    .catch(function(e) {

        function calculateAreaFromGeometry(geometry) {
            if (!geometry) return null;
            
            try {
                var totalArea = 0;
                var geomType = geometry.getType();
                
                if (geomType === 'MultiPolygon') {
                    var polygons = geometry.getPolygons();
                    for (var i = 0; i < polygons.length; i++) {
                        var poly = polygons[i];
                        var coords = poly.getCoordinates()[0];
                        if (coords && coords.length >= 3) {
                            var transformedCoords = [];
                            for (var j = 0; j < coords.length; j++) {
                                try {
                                    var coord4326 = ol.proj.transform(coords[j], 'EPSG:3857', 'EPSG:4326');
                                    transformedCoords.push(coord4326);
                                } catch (e) {
                                    transformedCoords.push(coords[j]);
                                }
                            }
                            var area = ol.sphere.getArea(transformedCoords);
                            if (area > 0 && !isNaN(area)) {
                                totalArea += Math.abs(area);
                            }
                        }
                    }
                } else if (geomType === 'Polygon') {
                    var coords = geometry.getCoordinates()[0];
                    if (coords && coords.length >= 3) {
                        var transformedCoords = [];
                        for (var j = 0; j < coords.length; j++) {
                            try {
                                var coord4326 = ol.proj.transform(coords[j], 'EPSG:3857', 'EPSG:4326');
                                transformedCoords.push(coord4326);
                            } catch (e) {
                                transformedCoords.push(coords[j]);
                            }
                        }
                        totalArea = ol.sphere.getArea(transformedCoords);
                        if (totalArea < 0) totalArea = Math.abs(totalArea);
                    }
                }
                
                return totalArea > 0 ? totalArea : null;
            } catch (e) {
                return null;
            }
        }

        // Function to find atoll from island_names table based on island name
        // Returns Atoll code (e.g., "AA", "ADh", "K") for filtering residents
        // Uses the same method as Boundaries popup
        function findAtollFromIslandName(islandName) {
            // Check both global variable and window object
            var islandFeatures = allIslandNameFeatures || window.allIslandNameFeatures;
            
            if (!islandName || !islandFeatures || islandFeatures.length === 0) {
                return '';
            }
            
            // Normalize island name for comparison (case-insensitive, trim)
            var normalizedIslandName = islandName.trim().toLowerCase();
            
            // Search through all island name features
            for (var i = 0; i < islandFeatures.length; i++) {
                var islandFeature = islandFeatures[i];
                var featureIslandName = islandFeature.get('IslandName') || '';
                
                // Compare normalized island names
                if (featureIslandName.trim().toLowerCase() === normalizedIslandName) {
                    // Found matching island - return its Atoll code
                    var atoll = islandFeature.get('Atoll') || '';
                    if (atoll) {
                        return atoll;
                    }
                }
            }
            
            return '';
        }

        // Function to find atoll name for an address by checking which administrative atoll contains it
        // Returns CodeName from AdministrativeAtoll_21.js as fallback
        function findAtollForAddress(addressFeature) {
            if (!addressFeature || !allAdminAtollFeatures || allAdminAtollFeatures.length === 0) {
                return '';
            }
            
            var addrCoord = addressFeature.getGeometry().getCoordinates();
            if (!addrCoord) return '';
            
            // Check which administrative atoll polygon contains this address point
            for (var i = 0; i < allAdminAtollFeatures.length; i++) {
                var atollFeature = allAdminAtollFeatures[i];
                var atollGeom = atollFeature.getGeometry();
                if (!atollGeom) continue;
                
                try {
                    if (atollGeom.intersectsCoordinate(addrCoord)) {
                        // Found the atoll - return CodeName (priority) for database filtering
                        return atollFeature.get('CodeName') || atollFeature.get('Name_Engli') || atollFeature.get('name') || atollFeature.get('Name') || '';
                    }
                } catch (e) {
                    try {
                        var extent = atollGeom.getExtent();
                        if (ol.extent.containsCoordinate(extent, addrCoord)) {
                            // Found the atoll - return CodeName (priority) for database filtering
                            return atollFeature.get('CodeName') || atollFeature.get('Name_Engli') || atollFeature.get('name') || atollFeature.get('Name') || '';
                        }
                    } catch (e2) {}
                }
            }
            
            return '';
        }

        function fetchResidentsForAddresses(addresses, category, shapeArea, calculatedArea, sqft, individualPlots) {
            if (addresses.length === 0) {
                showParcelInfo(category, shapeArea, calculatedArea, sqft, [], [], individualPlots);
                return;
            }
            
            var allResidents = [];
            var completedRequests = 0;
            var totalRequests = addresses.length;
            
            if (totalRequests === 0) {
                showParcelInfo(category, shapeArea, calculatedArea, sqft, addresses, [], individualPlots);
                return;
            }
            
            addresses.forEach(function(addr) {
                // Extract house name and island from Address_24_filtered.js
                var houseName = addr.name || ''; // Already extracted from address feature
                var island = '';
                var atoll = '';
                
                if (addr.feature) {
                    // Get island name from address feature
                    island = addr.feature.get('IslandName') || addr.feature.get('island') || '';
                    
                    // Try to get atoll directly from address feature first
                    atoll = addr.feature.get('Atoll') || addr.feature.get('atoll') || '';
                    
                    // If no atoll in address, look it up from IslandName_23.js
                    if (!atoll && island) {
                        atoll = findAtollFromIslandName(island);
                    }
                }
                
                // Build query parameters
                var params = new URLSearchParams();
                params.append('action', 'get_residents_by_address');
                params.append('address', addr.name);
                if (island) {
                    params.append('island', island);
                }
                if (atoll) {
                    params.append('atoll', atoll);
                }
                
                fetch('search-residents.php?' + params.toString())
                    .then(function(response) {
                        if (!response.ok) {
                            throw new Error('HTTP error: ' + response.status);
                        }
                        return response.json();
                    })
                    .then(function(data) {
                        if (data.success && data.residents) {
                            allResidents = allResidents.concat(data.residents);
                        }
                        completedRequests++;
                        
                        if (completedRequests === totalRequests) {
                            showParcelInfo(category, shapeArea, calculatedArea, sqft, addresses, allResidents, individualPlots);
                        }
                    })
                    .catch(function(error) {
                        console.error('Error fetching residents:', error);
                        completedRequests++;
                        if (completedRequests === totalRequests) {
                            showParcelInfo(category, shapeArea, calculatedArea, sqft, addresses, allResidents, individualPlots);
                        }
                    });
            });
        }

        // Function to calculate age from date of birth
        function calculateAge(dob) {
            if (!dob) return null;
            
            try {
                // Try to parse various date formats
                var birthDate = new Date(dob);
                
                // Check if date is valid
                if (isNaN(birthDate.getTime())) {
                    // Try parsing as YYYY-MM-DD or DD-MM-YYYY
                    var parts = dob.split(/[-\/]/);
                    if (parts.length === 3) {
                        // Try YYYY-MM-DD first
                        birthDate = new Date(parts[0], parts[1] - 1, parts[2]);
                        if (isNaN(birthDate.getTime())) {
                            // Try DD-MM-YYYY
                            birthDate = new Date(parts[2], parts[1] - 1, parts[0]);
            }
                    }
                }
                
                if (isNaN(birthDate.getTime())) {
                    return null;
                }
                
                var today = new Date();
                var age = today.getFullYear() - birthDate.getFullYear();
                var monthDiff = today.getMonth() - birthDate.getMonth();
                
                // Adjust age if birthday hasn't occurred this year
                if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birthDate.getDate())) {
                    age--;
                }
                
                return age >= 0 ? age : null;
            } catch (e) {
                return null;
            }
        }

            if (loadMoreIndicator) {
                loadMoreIndicator.innerHTML = '<i class="fa-solid fa-spinner fa-spin" style="margin-right:8px;"></i>Loading more residents...';
            }
            
            // Simulate slight delay for smooth UX
            setTimeout(function() {
                var nextBatch = allResidentsData.slice(displayedResidentsCount, displayedResidentsCount + residentsPerPage);
                
                nextBatch.forEach(function(resident) {
                    var cardHtml = renderResidentCard(resident);
                    residentsContainer.insertAdjacentHTML('beforeend', cardHtml);
                });
                
                displayedResidentsCount += nextBatch.length;

                    } else {
                        var remaining = allResidentsData.length - displayedResidentsCount;
                        loadMoreIndicator.innerHTML = 'Scroll for more residents... (' + remaining + ' remaining)';
                    }
                }
                
                isLoadingMore = false;
            }, 100);

        function performSearch(query, resultsContainer) {
            if (!query || query.length < 2) {
                if (resultsContainer) {
                    resultsContainer.classList.remove('visible');
                    resultsContainer.innerHTML = '';
                }
                return;
            }
            
            if (resultsContainer === searchResults) {
                resultsContainer.innerHTML = '<div style="padding: 20px; text-align: center; color: var(--text-muted);"><i class="fa-solid fa-spinner fa-spin" style="margin-right:8px;"></i>Searching...</div>';
                resultsContainer.classList.add('visible');
            } else {
                resultsContainer.innerHTML = '<div style="padding: 20px; text-align: center; color: var(--text-muted);"><i class="fa-solid fa-spinner fa-spin" style="margin-right:8px;"></i>Searching...</div>';
            }
            
            fetch('search-residents.php?action=search&q=' + encodeURIComponent(query))
                .then(function(response) {
                    if (!response.ok) {
                        return response.text().then(function(text) {
                            try {
                                var errorData = JSON.parse(text);
                                throw new Error(errorData.message || 'HTTP error: ' + response.status);
                            } catch (e) {
                                if (e.message && e.message.includes('HTTP error')) throw e;
                                throw new Error('HTTP error: ' + response.status);
                            }
                        });
                    }
                    return response.text().then(function(text) {
                        try {
                            return JSON.parse(text);
                        } catch (e) {
                            throw new Error('Invalid JSON response');
                        }
                    });
                })
                .then(function(data) {
                    if (data && data.success && data.results && data.results.length > 0) {
                        displaySearchResults(data.results, resultsContainer);
                    } else {
                        var errorMsg = (data && data.message) ? data.message : 'No results found';
                        resultsContainer.innerHTML = '<div style="padding: 20px; text-align: center; color: var(--text-muted);">' + errorMsg + '</div>';
                    }
                })
                .catch(function(error) {
                    console.error('Search error:', error);
                    resultsContainer.innerHTML = '<div style="padding: 20px; text-align: center; color: var(--danger-color);">Error: ' + (error.message || 'Search failed') + '</div>';
                });
        }

        window.zoomToSearchResult = function(resultIndex, element) {
            if (searchResults) searchResults.classList.remove('visible');
            if (searchInput) searchInput.value = '';
            closeMobileSearch();
            
            // Get result data
            var resultData = null;
            if (element && element.getAttribute('data-result')) {
                try {
                    resultData = JSON.parse(decodeURIComponent(element.getAttribute('data-result')));
                } catch (e) {
                    console.error('Error parsing result data:', e);
                }
            }
            
            if (!resultData && window.lastSearchResults && window.lastSearchResults[resultIndex]) {
                var result = window.lastSearchResults[resultIndex];
                resultData = {
                    address: result.permanent_address || result.address || '',
                    island: result.island || '',
                    atoll: result.atoll || '',
                    name: result.full_name || result.name || '',
                    idNo: result.id_no || result.id_number || ''
                };
            }
            
            if (!resultData || !resultData.address) {
                alert('Invalid search result data.');
                return;
            }
            
            function tryZoomToAddress() {
                if (!window.map) {
                    setTimeout(tryZoomToAddress, 500);
                    return;
                }
                
                // Use permanent_address as house_name (hname) to query address table directly
                var houseName = resultData.address.trim(); // permanent_address from search result
                var searchIsland = (resultData.island || '').trim();
                
                // Get atoll from island_names table (same method as Boundaries popup)
                var searchAtoll = '';
                if (searchIsland) {
                    searchAtoll = findAtollFromIslandName(searchIsland);
                }
                
                // If atoll not found from island_names, try using the one from search result as fallback
                if (!searchAtoll && resultData.atoll) {
                    searchAtoll = resultData.atoll.trim();
                }
                
                // Query address table directly using API with filters
                var format = new ol.format.GeoJSON();
                
                // Try with atoll filter first, then without if no results
                function tryFetchAddress(includeAtoll) {
                    var url = API_BASE_URL + '?layer=addresses';
                    if (houseName) {
                        url += '&hname=' + encodeURIComponent(houseName);
                    }
                    if (searchIsland) {
                        url += '&island=' + encodeURIComponent(searchIsland);
                        }
                    if (includeAtoll && searchAtoll) {
                        url += '&atoll=' + encodeURIComponent(searchAtoll);
                    }
                    
                    return fetch(url)
                        .then(function(r) { return r.json(); })
                        .then(function(data) {
                            if (data.features && data.features.length > 0) {
                                return data;
                            }
                            return null;
                        });
                }
                
                // Try with atoll first
                tryFetchAddress(true)
                    .then(function(data) {
                        // If no results with atoll, try without atoll
                        if (!data || !data.features || data.features.length === 0) {
                            if (searchAtoll) {
                                return tryFetchAddress(false);
                            }
                            return null;
                        }
                        return data;
                    })
                    .then(function(data) {
                        var foundAddress = null;
                            
                        if (data && data.features && data.features.length > 0) {
                            // Convert to OpenLayers features
                            var fetchedFeatures = format.readFeatures(data, {
                                dataProjection: 'EPSG:4326',
                                featureProjection: 'EPSG:3857'
                            });
                            
                            // Use the first result (should be exact match with filters)
                            if (fetchedFeatures.length > 0) {
                                foundAddress = fetchedFeatures[0];
                                
                                // Add to address source if not already there
                                if (USE_SQL_LAYERS && window.addressSource) {
                                    var alreadyAdded = false;
                                    window.addressSource.forEachFeature(function(f) {
                                        if (f.get('hname') === foundAddress.get('hname') && 
                                            f.get('IslandName') === foundAddress.get('IslandName')) {
                                            alreadyAdded = true;
                                            foundAddress = f; // Use existing feature
                                            return;
                            }
                        });
                                    if (!alreadyAdded) {
                                        window.addressSource.addFeature(foundAddress);
                                        if (!window.allAddressFeatures) {
                                            window.allAddressFeatures = [];
                            }
                                        window.allAddressFeatures.push(foundAddress);
                                    }
                                }
                            }
                    }
                    
                    if (foundAddress) {
                        var coord = foundAddress.getGeometry().getCoordinates();
                        window.map.getView().animate({
                            center: coord,
                            zoom: 19,
                            duration: 1000
                        });
                        
                        setTimeout(function() {
                            findAndShowParcelForAddress(foundAddress, coord);
                        }, 1200);
                    } else {
                            // Fallback: try to zoom to island center
                            if (searchIsland && window.allIslandNameFeatures) {
                                for (var i = 0; i < window.allIslandNameFeatures.length; i++) {
                                    var islandFeature = window.allIslandNameFeatures[i];
                                    var islandName = (islandFeature.get('IslandName') || '').trim();
                                    if (islandName.toLowerCase() === searchIsland.toLowerCase()) {
                                        var islandCoord = islandFeature.getGeometry().getCoordinates();
                                        window.map.getView().animate({
                                            center: islandCoord,
                                            zoom: 18,
                                            duration: 1000
                                        });
                                        setTimeout(function() {
                                            alert('Address "' + houseName + '" not found on map for ' + resultData.island + '. Zoomed to island. Please zoom in and try clicking the address on the map.');
                                        }, 1200);
                                        return;
                                    }
                                }
                            }
